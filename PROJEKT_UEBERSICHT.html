Spannungsmessung mit Oversampling (adc_reader.cpp)

static float read_channel(uint8_t channel) {
    pca9555_set_asel(channel);           // Multiplexer umschalten
    delayMicroseconds(ADC_SETTLING_US); // 100 µs Einschwingen
    uint32_t sum = 0;
    for (int i = 0; i < ADC_OVERSAMPLING; i++)  // 16 Samples
        sum += analogRead(PIN_ADC);
    return ((float)sum / ADC_OVERSAMPLING) * (ADC_VREF / ADC_RESOLUTION);
}
// SOC: LUT-Interpolation + EMA-Filter (α = 0.1)

Interrupt-sicheres I²C (pca9555_module.cpp)

// ISR – nur Signal geben, kein I²C:
static void IRAM_ATTR pca9555_isr() {
    BaseType_t woken = pdFALSE;
    xSemaphoreGiveFromISR(_irq_sem, &woken);
    portYIELD_FROM_ISR(woken);
}
// FreeRTOS-Task erledigt I²C-Kommunikation thread-safe:
static void pca9555_irq_task(void* arg) {
    for (;;) {
        xSemaphoreTake(_irq_sem, portMAX_DELAY);   // warten
        xSemaphoreTake(_wire_mutex, portMAX_DELAY); // I²C sperren
        // Ports lesen, dekodieren, Callback aufrufen
        xSemaphoreGive(_wire_mutex);
    }
}

Automatik STATE_OFF → Übergang (state_machine.cpp)

case STATE_OFF:
    if (mode == MODE_AUTO) {
        if (s.soc >= THR_SOC_MIN && !s.grid_ok && s.v_load < THR_SWITCH_V)
            enter_state(STATE_TRANSITION_TO_ISLAND); // Batterie ok, kein Netz
        else if (s.soc < THR_SOC_MAX && s.grid_ok && s.v_load < THR_SWITCH_V)
            enter_state(STATE_TRANSITION_TO_GRID);   // Netz ok, Bat. nicht voll
    }
    break;
